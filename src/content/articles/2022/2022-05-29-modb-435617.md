构建实体类

```
package com.zuoyang.springbootmongo.domain;


import org.springframework.data.mongodb.core.mapping.Document;


@Document(collection = "hero")
public class Hero {
private String name;
private String gender;
private Integer age;
private long dt;


public String getName() {
return name;
    }


public void setName(String name) {
this.name = name;
    }


public String getGender() {
return gender;
    }


public void setGender(String gender) {
this.gender = gender;
    }


public Integer getAge() {
return age;
    }


public void setAge(Integer age) {
this.age = age;
    }


public long getDt() {
return dt;
    }


public void setDt(long dt) {
this.dt = dt;
    }
}
```

# mongo数据链接
test_client = pymongo.MongoClient("mongodb://admin:123456@192.0.2.13:27017/?authSource=admin")

**Golang示例**

初始化工程目录

```
go mod init
```

下载MongoDB连接操作包

```
go get go.mongodb.org/mongo-driver/mongo
```

示例代码

```
package main


import (
"context"
"fmt"
"go.mongodb.org/mongo-driver/mongo"
"go.mongodb.org/mongo-driver/mongo/options"
"log"
"time"
)


type hero struct{
  Name string
  Gender string
  Age int
  Dt int64
}
var client *mongo.Client


func main() {
// mongo数据链接
    clientOptions := options.Client().ApplyURI("mongodb://admin:123456@192.0.2.13:27017/?authSource=admin")
// 连接到MongoDB
var err error
    client, err = mongo.Connect(context.TODO(), clientOptions)
if err != nil {
        log.Fatal(err)
    }
// 测试连接
    err = client.Ping(context.TODO(), nil)
if err != nil {
        log.Fatal(err)
    }
    fmt.Println("连接成功")
//插入测试
  chen:=hero{"陈","男",55,time.Now().UnixNano()}
  fmt.Println(chen)
// client.Database("数据库名").Collection("集合名称")
  collection:=client.Database("zuoyang").Collection("hero")
  x,err:=collection.InsertOne(context.TODO(),chen)
if err != nil {
        log.Fatal(err)
    }
  fmt.Println(x.InsertedID)
// golang获取时间戳 毫秒级
  fmt.Println(time.Now().UnixNano())
}
```
