```
package com.zuoyang.execl_demo.enity;
import cn.afterturn.easypoi.excel.annotation.Excel;
import java.util.Date;
public class Hero {
//    @Excel(name = "编号", orderNum = "0")
    private String id;
    @Excel(name = "姓名", orderNum = "0")

    @Excel(name = "性别", orderNum = "1")
    private String gender;
    @Excel(name = "职业", orderNum = "2")
    private String position;
    @Excel(name = "备注", orderNum = "3" ,width = 35,height = 40)
    private String note;
    @Excel(name = "年龄", orderNum = "4")
    private int age;
    @Excel(name = "出生时间", width = 25, format = "yyyy-MM-dd HH:mm:ss",orderNum = "5")
    private Date createDate;
    public int getAge() {
        return age;
    }
    public void setAge(int age) {
        this.age = age;
    }
    public String getId() {
        return id;
    }
    public void setId(String id) {
        this.id = id;
    }
    public String getName() {
        return name;
    }
    public void setName(String name) {
        this.name = name;
    }
    public String getGender() {
        return gender;
    }
    public void setGender(String gender) {
        this.gender = gender;
    }
    public String getPosition() {
        return position;
    }
    public void setPosition(String position) {
        this.position = position;
    }
    public String getNote() {
        return note;
    }
    public void setNote(String note) {
        this.note = note;
    }
    public Date getCreateDate() {
        return createDate;
    }
    public void setCreateDate(Date createDate) {
        this.createDate = createDate;
    }
    @Override

        return "Hero{" +

                ", name='" + name + '\'' +

                ", position='" + position + '\'' +
                ", note='" + note + '\'' +
                ", age=" + age +
                ", createDate=" + createDate +
                '}';
    }
}
```

**控制器类**

```
package com.zuoyang.execl_demo.rest;
import cn.afterturn.easypoi.excel.ExcelExportUtil;
import cn.afterturn.easypoi.excel.ExcelImportUtil;
import cn.afterturn.easypoi.excel.entity.ExportParams;
import cn.afterturn.easypoi.excel.entity.ImportParams;
import com.zuoyang.execl_demo.enity.Hero;
import org.apache.poi.ss.usermodel.Workbook;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;
import org.springframework.web.multipart.MultipartFile;
import javax.servlet.http.HttpServletResponse;
import java.io.IOException;
import java.net.URLEncoder;
import java.util.ArrayList;
import java.util.Date;
import java.util.List;
@RestController
public class ExeclHeroController {
    /**
     * 下载模板
     *
     * @param response
     */
    @RequestMapping("getTemplate")
    public void getTemplate(HttpServletResponse response) {
        List<Hero> heroList = new ArrayList<>();
        Workbook workbook = ExcelExportUtil.exportExcel(new ExportParams("元末明初英雄信息", "明朝英雄"), Hero.class, heroList);
        downLoadExcel("Hero.xls", response, workbook);
    }
    /**
     * 数据导出
     *
     * @param response
     */
    @RequestMapping("exportExcel")
    public void export(HttpServletResponse response) {
        List<Hero> heroList = new ArrayList<>();
        Hero yaoguangxiao = new Hero();
        yaoguangxiao.setId("1");
        yaoguangxiao.setName("姚广孝");
        yaoguangxiao.setGender("男");
        yaoguangxiao.setPosition("谋士");
        yaoguangxiao.setNote("给朱元璋和朱棣打工的");
        yaoguangxiao.setAge(57);
        yaoguangxiao.setCreateDate(new Date());
        Hero liubowen = new Hero();
        liubowen.setId("2");
        liubowen.setName("刘伯温");
        liubowen.setGender("男");
        liubowen.setPosition("谋士");
        liubowen.setNote("给朱元璋和朱棣打工的");
        liubowen.setAge(55);
        liubowen.setCreateDate(new Date());
        heroList.add(yaoguangxiao);
        heroList.add(liubowen);
        //导出操作
        Workbook workbook = ExcelExportUtil.exportExcel(new ExportParams("元末明初英雄信息", "明朝英雄"), Hero.class, heroList);
        downLoadExcel("Hero.xls", response, workbook);
    }
    private static void downLoadExcel(String fileName, HttpServletResponse response, Workbook workbook) {
        try {
            response.setCharacterEncoding("UTF-8");
            response.setHeader("content-Type", "application/vnd.ms-excel");
            response.setHeader("Content-Disposition",
                    "attachment;filename=" + URLEncoder.encode(fileName, "UTF-8"));
            workbook.write(response.getOutputStream());
        } catch (IOException e) {
            throw new RuntimeException(e.getMessage());
        }
    }
    /**
     * 数据导入
     *
     * @param file
     * @return
     * @throws Exception
     */
    @RequestMapping("importExcel")
    public String importExcel(MultipartFile file) throws Exception {
        ImportParams params = new ImportParams();
        params.setTitleRows(1);
        params.setHeadRows(1);
        List<Hero> heroList = ExcelImportUtil.importExcel(file.getInputStream(), Hero.class, params);
        heroList.stream().forEach(System.out::println);
        return "ok";
    }
}
```
