```
package com.zy.nettysocket.nettyserver;
import io.netty.channel.ChannelHandlerContext;
import io.netty.channel.SimpleChannelInboundHandler;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.data.redis.core.RedisTemplate;
public class ServerHandler extends SimpleChannelInboundHandler<String> {




    @Override
    protected void channelRead0(ChannelHandlerContext ctx, String msg) throws   Exception {
//        处理服务端异常
        //接收到的数据
        System.out.println("客户端数据： "+msg);
        //返回给客户端的数据
        ctx.channel().writeAndFlush(msg);
//        关闭通道连接
        ctx.close();
    }
    @Override
    public void exceptionCaught(ChannelHandlerContext ctx, Throwable cause) throws  Exception {
//        此处捕获连接中的异常
        cause.printStackTrace();
        ctx.close();
    }
}
```

ServerInitializer 通道初始化处理器

```
package com.zy.nettysocket.nettyserver;
import io.netty.channel.ChannelInitializer;
import io.netty.channel.ChannelPipeline;
import io.netty.channel.socket.SocketChannel;
import io.netty.handler.codec.LengthFieldBasedFrameDecoder;
import io.netty.handler.codec.LengthFieldPrepender;
import io.netty.handler.codec.string.StringDecoder;
import io.netty.handler.codec.string.StringEncoder;
import io.netty.util.CharsetUtil;
public class ServerInitializer extends ChannelInitializer<SocketChannel> {
    @Override
    protected void initChannel(SocketChannel ch) throws Exception {
        ChannelPipeline pipeline = ch.pipeline();
        //数据编码 数据分包，组包，粘包
        pipeline.addLast(new LengthFieldBasedFrameDecoder(Integer.MAX_VALUE,0,4,0,4));
        pipeline.addLast(new LengthFieldPrepender(4));
//        转换为字符串
        pipeline.addLast(new StringDecoder(CharsetUtil.UTF_8));
//        转换为字节码
        pipeline.addLast(new StringEncoder(CharsetUtil.UTF_8));
        pipeline.addLast(new ServerHandler());




    }
}
```

ServerSocket

```
package com.zy.nettysocket.nettyserver;
import io.netty.bootstrap.ServerBootstrap;
import io.netty.channel.ChannelFuture;
import io.netty.channel.nio.NioEventLoopGroup;
import io.netty.channel.socket.nio.NioServerSocketChannel;
public class ServerSocket {
    public static void main(String ...arg) throws Exception {
        //负责接收客户端连接池
        NioEventLoopGroup bossGroup = new NioEventLoopGroup();
        //负责处理连接池
        NioEventLoopGroup wokerGroup = new NioEventLoopGroup();
        try{
            ServerBootstrap bootstrap = new ServerBootstrap();
            bootstrap.group(bossGroup,wokerGroup)
                    .channel(NioServerSocketChannel.class)
                    .childHandler(new ServerInitializer());
            //绑定端口号
            ChannelFuture channelFuture = bootstrap.bind(9999).sync();
            channelFuture.channel().closeFuture().sync();
        } finally {
            bossGroup.shutdownGracefully();
            wokerGroup.shutdownGracefully();
        }
    }
}
```

```
package com.zy.nettysocket.nettyclient;
import io.netty.channel.ChannelHandlerContext;
import io.netty.channel.SimpleChannelInboundHandler;
import java.time.LocalDateTime;
public class ClientHandler extends SimpleChannelInboundHandler<String> {
    //接收服务端数据&发送数据
    @Override
    protected void channelRead0(ChannelHandlerContext ctx, String msg) throws Exception {
//        处理客户端业务逻辑
//        接受服务端数据
        System.out.println(msg);
//      向服务端发送数据
        ctx.writeAndFlush(LocalDateTime.now());
        //完成通信后关闭连接
        ctx.close();
    }
    //和服务器建立连接
    @Override
    public void channelActive(ChannelHandlerContext ctx) throws Exception {
//        向服务发送信息
        ctx.writeAndFlush("hello netty !");
    }
    @Override
    public void exceptionCaught(ChannelHandlerContext ctx, Throwable cause) throws Exception {
        //        此处捕获连接中的异常
        cause.printStackTrace();
        ctx.close();
    }
}
```

```
import io.netty.handler.codec.LengthFieldPrepender;
```

```
package com.zy.nettysocket.nettyclient;
import io.netty.bootstrap.Bootstrap;
import io.netty.channel.ChannelFuture;
import io.netty.channel.nio.NioEventLoopGroup;
import io.netty.channel.socket.nio.NioSocketChannel;
public class ClientSocket {
    public static void main(String[] arg) throws Exception {
//        客户端连接池
        NioEventLoopGroup eventLoopGroup = new NioEventLoopGroup();
        try {
            Bootstrap bootstrap = new Bootstrap();
            bootstrap.group(eventLoopGroup).channel(NioSocketChannel.class)
                    .handler(new Clientinitializer());
//          向服务端发起连接，IP localhost,PORT 9999
            ChannelFuture channelFuture = bootstrap.connect("localhost", 9999).sync();
//            保持通道连接
            channelFuture.channel().closeFuture().sync();
        } finally {
            eventLoopGroup.shutdownGracefully();
        }
    }
}
```

```
package com.zy.nettysocket.config;//package com.zy.nettysocket.socketss;
import com.zy.nettysocket.thread.ServerSocketThread;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.context.annotation.Bean;
import org.springframework.core.env.Environment;
import org.springframework.stereotype.Component;
import java.util.concurrent.LinkedBlockingQueue;
import java.util.concurrent.ThreadPoolExecutor;
import java.util.concurrent.TimeUnit;
@Component
public class ServerSocketConfig {
    private static Logger log = LoggerFactory.getLogger(ServerSocketConfig.class);
    private static final ThreadPoolExecutor threadpool = new ThreadPoolExecutor(1, 1,
            10L, TimeUnit.SECONDS, new LinkedBlockingQueue<Runnable>());
    @Autowired
    private Environment environment;
    @Bean
    public void socketCreate() {
        log.info("socket服务端开启");
        String socketServerPort = environment.getProperty("socket-server-port");
        threadpool.execute(new ServerSocketThread(socketServerPort));
    }
}
```

ServerSocketThread

```
package com.zy.nettysocket.thread;
import com.zy.nettysocket.handler.ServerInitializer;
import io.netty.bootstrap.ServerBootstrap;
import io.netty.channel.ChannelFuture;
import io.netty.channel.nio.NioEventLoopGroup;
import io.netty.channel.socket.nio.NioServerSocketChannel;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
public class ServerSocketThread implements Runnable {
    private String socketServerPort;
    private static Logger log = LoggerFactory.getLogger(ServerSocketThread.class);
    public ServerSocketThread(String socketServerPort) {
        this.socketServerPort = socketServerPort;
    }
    @Override
    public void run() {
        //负责接收客户端连接
        NioEventLoopGroup bossGroup = new NioEventLoopGroup();
        //负责处理连接
        NioEventLoopGroup wokerGroup = new NioEventLoopGroup();
        try {
            ServerBootstrap bootstrap = new ServerBootstrap();
            bootstrap.group(bossGroup, wokerGroup)
                    .channel(NioServerSocketChannel.class)
                    .childHandler(new ServerInitializer());
            //绑定端口号
            ChannelFuture channelFuture = bootstrap.bind(Integer.parseInt(socketServerPort)).sync();
            channelFuture.channel().closeFuture().sync();
            log.info("socket server start success, server port {}",socketServerPort);
        } catch (InterruptedException e) {
            log.error("socket server start fail, error msg {}",e);
        } finally {
            bossGroup.shutdownGracefully();
            wokerGroup.shutdownGracefully();
        }
    }
}
```
