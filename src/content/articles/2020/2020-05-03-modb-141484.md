```
import lombok.extern.slf4j.Slf4j;
import org.bytedeco.javacv.FFmpegFrameGrabber;
import org.bytedeco.javacv.Frame;
import org.bytedeco.javacv.FrameGrabber;
import org.bytedeco.javacv.Java2DFrameConverter;




import javax.imageio.ImageIO;
import java.awt.image.BufferedImage;
import java.io.ByteArrayOutputStream;
import java.io.File;
import java.io.IOException;




/**
 * @program: pigx
 * @description: Java获取视频帧
 * @author: 左羊
 * @create: 2020-05-03 15:29
 **/
@Slf4j
public class CVDemo {




   /**
    * 获取视频帧
    * @param type
    */
   public static void getFrame(String type) {
      //Frame对象,帧对象
      Frame frame = null;
      /**
       * HTTP协议直播源
       * CCTV1高清：http://ivi.bupt.edu.cn/hls/cctv1hd.m3u8
       * CCTV3高清：http://ivi.bupt.edu.cn/hls/cctv3hd.m3u8
       * CCTV5高清：http://ivi.bupt.edu.cn/hls/cctv5hd.m3u8
       * CCTV5+高清：http://ivi.bupt.edu.cn/hls/cctv5phd.m3u8
       * CCTV6高清：http://ivi.bupt.edu.cn/hls/cctv6hd.m3u8
       */
      String file = "http://ivi.bupt.edu.cn/hls/cctv1hd.m3u8";
      FFmpegFrameGrabber fFmpegFrameGrabber = null;
      FrameGrabber grabber = null;
      try {
         int width = 860, height = 480;
         if ("1".equals(type)) {
            fFmpegFrameGrabber = FFmpegFrameGrabber.createDefault(file);
            fFmpegFrameGrabber.setOption("rtsp_transport", "tcp");
//       设置帧率
            fFmpegFrameGrabber.setFrameRate(6);
            fFmpegFrameGrabber.setImageWidth(width);
            fFmpegFrameGrabber.setImageHeight(height);
         } else {
            grabber = FrameGrabber.createDefault(0);
            grabber.setOption("rtsp_transport", "tcp");
            grabber.setFrameRate(6);
            System.out.println("帧率:" + grabber.getFrameRate());
            System.out.println("像素格式为：" + grabber.getPixelFormat());
            grabber.setImageWidth(width);
            grabber.setImageHeight(height);
         }
      } catch (FrameGrabber.Exception e) {
         e.printStackTrace();
      }
      try {
         if ("1".equals(type)) {
            fFmpegFrameGrabber.start();
         } else {
            grabber.start();
         }




         long count = 0L;
         while (true) {
            //文件储存对象
            File outPut = new File("E://image/"+count+".png");
            //获取帧
            if ("1".equals(type)) {
               frame = fFmpegFrameGrabber.grabImage();
            } else {
               frame = grabber.grabFrame();
            }
            if (frame != null) {
               ImageIO.write(FrameToBufferedImage(frame), "png", outPut);
               count++;
            }
         }
      } catch (IOException e) {
         log.error(e.getMessage());
      }
   }




   /**
    * 帧转图片对象
    * @param frame
    * @return
    */
   public static BufferedImage FrameToBufferedImage(Frame frame) {
      //创建BufferedImage对象
      Java2DFrameConverter converter = new Java2DFrameConverter();
      BufferedImage bufferedImage = converter.getBufferedImage(frame);
      return bufferedImage;
   }
   public static void main(String[] args) {
      getFrame("0");
 }
```

帧对象Frame，由JavaCV提供的储存帧对象。

视频流抓取类FFmpegFrameGrabber，用于以Ffmpeg方式抓取网络或本地音视频对象。

视频流抓取类FrameGrabber，用于获取本机音视频设备抓取音视频流。

```
fFmpegFrameGrabber = FFmpegFrameGrabber.createDefault(file);
用于与网络或音视频文件建立连接
grabber = FrameGrabber.createDefault(0);
获取本地音视频设备连接
```

基础配置

```
fFmpegFrameGrabber.setOption("rtsp_transport", "tcp");
设置传输方式，如果是网络摄像头或网络流，我建议以TCP的方式建立连接，不然JavaCV默认的传输方式为UDP，会出现丢包的现象。
fFmpegFrameGrabber.setFrameRate(6);
设置帧率，表示没有看出什么效果。
fFmpegFrameGrabber.setImageWidth(width);
设置帧宽度也就是图片宽度
fFmpegFrameGrabber.setImageHeight(height);
设置帧高度也就是图片高度
fFmpegFrameGrabber.start();
grabber.start();
启动
frame = fFmpegFrameGrabber.grabImage();
frame = grabber.grabFrame();
获取帧
```
